<script>
document.addEventListener("DOMContentLoaded", function() {
  document.getElementById('ProductCategory').addEventListener('change', () => {
    const cat = document.getElementById('ProductCategory').value;
    document.getElementById('extractTypeGroup').style.display = (cat === 'Extracts') ? 'block' : 'none';
  });

  const imgInput = document.getElementById('ImageUpload');
  const previewRow = document.getElementById('previewRow');
  const spinner = document.getElementById('uploadSpinner');
  const imageUrlsField = document.getElementById('ImageUrls');
  let uploadedLinks = [];

  function updateImageUrls() {
    imageUrlsField.value = uploadedLinks.join(',');
  }

  imgInput.onchange = function(e) {
    const files = Array.from(e.target.files);
    if (!files.length) {
      alert('❌ No files selected.');
      return;
    }
    if (files.length + uploadedLinks.length > 5) {
      alert("Limit 5 images.");
      return;
    }
    spinner.style.display = "block";

    files.forEach((file) => {
      const reader = new FileReader();
      reader.onload = function(evt) {
        google.script.run
          .withSuccessHandler(function(resp) {
            spinner.style.display = "none";
            if (resp && resp.ok && resp.url) {
              uploadedLinks.push(resp.url);
              const wrap = document.createElement('div');
              wrap.className = 'preview-img';
              wrap.innerHTML = `
                <img src="${resp.url}">
                <button class="delbtn">✕</button>`;
              wrap.querySelector('.delbtn').onclick = function() {
                wrap.remove();
                uploadedLinks = uploadedLinks.filter(u => u !== resp.url);
                updateImageUrls();
              };
              previewRow.appendChild(wrap);
              updateImageUrls();
            } else {
              alert("Telegram upload failed: " + (resp.error || 'Unknown error'));
            }
          })
          .withFailureHandler(function(err) {
            spinner.style.display = "none";
            alert("Script error: " + (err.message || err));
          })
          .uploadPhotoToTelegramWithMeta(evt.target.result, {});
      };
      reader.readAsDataURL(file);
    });
  };

  document.getElementById('addInventoryForm').onsubmit = function(e) {
    e.preventDefault();
    const btn = document.getElementById('addBtn');
    const resultDiv = document.getElementById('result');
    btn.disabled = true;
    resultDiv.innerHTML = '';

    const cat = document.getElementById('ProductCategory').value;
    let type = "";
    if (cat === "Extracts") type = document.getElementById('Type').value;
    else if (cat === "Bulk Flower") type = "Bulk Flower";
    else if (cat === "Pre-Rolls") type = "Pre-Rolls";

    const data = {
      Handle: document.getElementById('Handle').value,
      Title: document.getElementById('Title').value,
      Vendor: document.getElementById('Vendor').value,
      ProductCategory: cat,
      Type: type,
      Weight: document.getElementById('Weight').value,
      Tier: document.getElementById('Tier').value,
      Quantity: document.getElementById('Quantity').value,
      Description: document.getElementById('Description').value,
      ImageUrl: imageUrlsField.value
    };

    google.script.run
      .withSuccessHandler(response => {
        btn.disabled = false;
        if (response && response.success) {
          resultDiv.innerHTML = '✅ Added!';
          document.getElementById('addInventoryForm').reset();
          previewRow.innerHTML = '';
          uploadedLinks = [];
          updateImageUrls();
        } else {
          resultDiv.innerHTML = '❌ Error: Could not add item.';
        }
      })
      .withFailureHandler(err => {
        btn.disabled = false;
        resultDiv.innerHTML = `❌ Script Error: ${err.message || err}`;
      })
      .addItem(data);
  };
});
</script>
