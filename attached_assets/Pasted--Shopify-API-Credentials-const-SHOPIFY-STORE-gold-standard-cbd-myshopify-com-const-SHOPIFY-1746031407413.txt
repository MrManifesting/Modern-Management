// 🔐 Shopify API Credentials
const SHOPIFY_STORE = 'gold-standard-cbd.myshopify.com';
const SHOPIFY_ACCESS_TOKEN = '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';

/**
 * 🔁 Sync SKUs & Inventory on 'THCa Flower' tab,
 * seed with Amount (col B), generate SKU (col L),
 * sync live qty (col M), and build Barcode|COA QR links (col O).
 */
function updateTHCaFlowerInventory() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('THCa Flower');
  if (!sheet) {
    SpreadsheetApp.getUi().alert("❌ Sheet 'THCa Flower' not found.");
    return;
  }

  // Prepare InventoryLog sheet
  let logSheet = ss.getSheetByName('InventoryLog');
  if (!logSheet) logSheet = ss.insertSheet('InventoryLog');
  if (logSheet.getLastRow() === 0) {
    logSheet.appendRow(['Timestamp','Strain','SKU','Previous Qty','Updated Qty','Status']);
  }

  const startRow     = 7;    // data starts on row 7
  const skuCol       = 12;   // column L
  const inventoryCol = 13;   // column M
  const tagCol       = 14;   // column N
  const linkCol      = 15;   // column O
  const lastRow      = sheet.getLastRow();
  const rows         = sheet.getRange(startRow, 1, lastRow - startRow + 1, linkCol).getValues();

  // Fetch Shopify products
  const resp     = UrlFetchApp.fetch(
    `https://${SHOPIFY_STORE}/admin/api/2023-10/products.json?limit=250`,
    { headers: { 'X-Shopify-Access-Token': SHOPIFY_ACCESS_TOKEN } }
  );
  const products = JSON.parse(resp.getContentText()).products;

  // Build SKU->quantity map and Tag->quantity map
  const skuMap = {}, tagMap = {};
  products.forEach(p => {
    p.variants.forEach(v => {
      if (v.sku) skuMap[v.sku.toUpperCase()] = v.inventory_quantity;
    });
    (p.tags||'').split(',').map(t=>t.trim().toLowerCase()).forEach(tag => {
      const q = p.variants[0]?.inventory_quantity;
      if (q != null) tagMap[tag] = Math.max(tagMap[tag]||0, q);
    });
  });

  // Process each row
  rows.forEach((r, i) => {
    const strain = r[0];             // A
    const amount = r[1];             // B
    const pct    = r[2];             // C
    const tier   = r[3];             // D
    const coaUrl = r[8];             // I
    let sku      = r[skuCol-1];      // L
    const tag    = (r[tagCol-1]||'').toLowerCase().trim();
    const prev   = r[inventoryCol-1] || '';
    const cellInv= sheet.getRange(startRow + i, inventoryCol);
    const cellLink=sheet.getRange(startRow + i, linkCol);

    // 1) Seed Inventory with Amount
    cellInv.setValue(amount);

    // 2) Generate SKU if missing
    if (!sku && strain && pct != null && tier) {
      const initials = strain.split(' ').map(w=>w[0]).join('').toUpperCase();
      const tcode    = tier.replace(/[^A-Z]/gi,'').substring(0,3).toUpperCase();
      const code     = Math.round(parseFloat(pct) * 10000);
      sku = `${initials}-${tcode}-${code}`;
      sheet.getRange(startRow + i, skuCol).setValue(sku);
    }
    if (!sku && !tag) return;

    // 3) Lookup live quantity
    let liveQty = skuMap[sku.toUpperCase()];
    if (liveQty == null && tag) liveQty = tagMap[tag];

    // 4) Override seeded amount if live qty available
    if (liveQty != null) {
      cellInv.setValue(liveQty).setBackground(liveQty < 10 ? '#ffcccc':'#ffffff');
      logSheet.appendRow([new Date(), strain, sku, prev, liveQty, 'UPDATED']);
    }

    // 5) Set clickable Barcode|COA QR in column O
    if (sku) {
      const barcodeUrl =
        'https://barcode.tec-it.com/barcode.ashx?data=' + encodeURIComponent(sku) + '&code=Code128&dpi=96';
      const qrUrl =
        'https://api.qrserver.com/v1/create-qr-code/?size=150x150&data=' + encodeURIComponent(coaUrl);
      const formula = `=HYPERLINK("${barcodeUrl}","Barcode") & " | " & HYPERLINK("${qrUrl}","COA QR")`;
      cellLink.setFormula(formula);
    }
  });
}

/**
 * 📋 Refresh InventoryList tab
 */
function refreshInventoryList() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let list = ss.getSheetByName('InventoryList');
  if (!list) list = ss.insertSheet('InventoryList');
  list.clear();
  list.appendRow(['Strain','Amount','%','Tier','Strain Type','SKU']);
  const sheet = ss.getSheetByName('THCa Flower');
  const data  = sheet.getRange(7,1,sheet.getLastRow()-6,12).getValues();
  data.forEach(r => {
    const [s,a,p,t,,,,,,type,sku] = r;
    if (!s) return;
    list.appendRow([s,a,p,t,type,sku]);
  });
}

/**
 * 🖨️ Generate printable labels in a Google Doc (Avery 5160)
 */
function generateBarcodesAndQRCodes() {
  const doc   = DocumentApp.create('THCa Flower Barcode & QR Labels');
  const body  = doc.getBody();
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('THCa Flower');
  if (!sheet) return;
  const data = sheet.getRange(7,1,sheet.getLastRow()-6,12).getValues();
  const perRow = 3;
  let rowArr = [];

  data.forEach(r => {
    const strain = r[0];
    const qty    = r[1];
    const coaUrl = r[8];
    const sku    = r[11];
    if (!strain || !sku) return;

    const barcodeBlob = UrlFetchApp.fetch(
      'https://barcode.tec-it.com/barcode.ashx?data=' + encodeURIComponent(sku) + '&code=Code128&dpi=96'
    ).getBlob();
    const qrBlob = UrlFetchApp.fetch(
      'https://api.qrserver.com/v1/create-qr-code/?size=150x150&data=' + encodeURIComponent(coaUrl)
    ).getBlob();
    const txt = strain + '\nSKU: ' + sku + '\nQty: ' + qty;

    rowArr.push([txt, barcodeBlob, qrBlob]);
    if (rowArr.length === perRow) {
      const tr = body.appendTable().appendTableRow();
      rowArr.forEach(([t,b,q]) => {
        const cell = tr.appendTableCell();
        cell.appendParagraph(t);
        cell.appendImage(b).setWidth(150);
        cell.appendImage(q).setWidth(100);
      });
      body.appendParagraph('\n');
      rowArr = [];
    }
  });
  if (rowArr.length) {
    const tr = body.appendTable().appendTableRow();
    rowArr.forEach(([t,b,q]) => {
      const cell = tr.appendTableCell();
      cell.appendParagraph(t);
      cell.appendImage(b).setWidth(150);
      cell.appendImage(q).setWidth(100);
    });
  }
  SpreadsheetApp.getUi().alert('✅ Labels built: ' + doc.getUrl());
}
function generateAveryLabelsDoc() { generateBarcodesAndQRCodes(); }

/**
 * 📤 Export Labels as PDF to Drive
 */
function exportLabelsAsPDF() {
  const name = 'THCa Flower Barcode & QR Labels';
  const files = DriveApp.getFilesByName(name);
  if (!files.hasNext()) {
    SpreadsheetApp.getUi().alert('❌ Doc "' + name + '" not found.');
    return;
  }
  const file = files.next();
  const pdf = file.getAs('application/pdf').setName(name + '.pdf');
  const saved = DriveApp.createFile(pdf);
  SpreadsheetApp.getUi().alert('✅ PDF: ' + saved.getUrl());
}

/**
 * 🔄 Sync inventory by Handle from ShopifyInventoryExport sheet
 */
function updateInventoryFromExport() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const exp = ss.getSheetByName('ShopifyInventoryExport');
  if (!exp) {
    SpreadsheetApp.getUi().alert("❌ Sheet 'ShopifyInventoryExport' not found.");
    return;
  }
  const data = exp.getRange(2,1,exp.getLastRow()-1,17).getValues();
  const map = {};
  data.forEach(r => {
    const h = r[0]; // Handle
    const avail = Number(r[15]) || 0; // Available
    const onHand = Number(r[16]) || 0;
    map[h] = avail > 0 ? avail : onHand;
  });
  const sheet = ss.getSheetByName('THCa Flower');
  const rows = sheet.getRange(7,16,sheet.getLastRow()-6,1).getValues();
  rows.forEach((r,i) => {
    const h = r[0];
    if (h && map[h]!=null) {
      sheet.getRange(7+i,13).setValue(map[h]);
    }
  });
  SpreadsheetApp.getUi().alert('✅ Inventory updated from CSV export.');
}

/**
 * Trigger on edit of THCa Flower
 */
function onEdit(e) {
  if (e.range.getSheet().getName() === 'THCa Flower') {
    updateTHCaFlowerInventory();
    refreshInventoryList();
  }
}

/**
 * Set up custom menu and auto-run on open
 */
function onOpen() {
  const ui = SpreadsheetApp.getUi();
  ui.createMenu('🛠 Shopify Sync')
    .addItem('📥 Sync Inventory + SKU','updateTHCaFlowerInventory')
    .addItem('🔄 Sync from CSV Export','updateInventoryFromExport')
    .addItem('🗂 Refresh Inventory List','refreshInventoryList')
    .addItem('🖨️ Generate Barcode Labels','generateBarcodesAndQRCodes')
    .addItem('📤 Export Labels as PDF','exportLabelsAsPDF')
    .addItem('📊 View Inventory Log','openInventoryLog')
    .addToUi();
  updateTHCaFlowerInventory();
  refreshInventoryList();
}

/**
 * Show InventoryLog sheet
 */
function openInventoryLog() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const log = ss.getSheetByName('InventoryLog') || ss.insertSheet('InventoryLog');
  ss.setActiveSheet(log);
}
