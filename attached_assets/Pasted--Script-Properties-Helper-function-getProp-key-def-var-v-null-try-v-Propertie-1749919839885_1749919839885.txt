// Your existing autoAssignSKUsUniversal function (keep as is)
function autoAssignSKUsUniversal() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  if (!data.length) return;
  var headers = data[0].map(h => (h || '').toString());
  function idx(col) { return headers.findIndex(h => h.trim().toLowerCase() === col.trim().toLowerCase()); }
  var handleCol = idx('Handle');
  var weightCol = idx('Option1 Value');
  var skuCol = idx('SKU');
  if (skuCol === -1) skuCol = idx('Variant SKU');
  if ([handleCol, weightCol, skuCol].some(i => i === -1)) {
    SpreadsheetApp.getUi().alert('❌ One or more columns not found!\nActual headers: ' + headers.join(', '));
    return;
  }
  var weightMap = {
    '1 pound': '454','28 grams': '28','14 grams': '14','7 grams': '7','3.5 grams': '3.5',
    '1 g': '1','1 gram': '1','1 lb': '454','pound': '454'
  };
  var updated = 0;
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    if (!row[skuCol]) {
      var handle = (row[handleCol] || '').toString().trim();
      var weight = (row[weightCol] || '').toString().toLowerCase().trim();
      var abbr = handle.split(/[^a-zA-Z0-9]+/).map(part => part.charAt(0)).join('').toUpperCase();
      var weightNum = weightMap[weight];
      if (weightNum === undefined) {
        var found = weight.match(/(\d+(\.\d+)?)/);
        weightNum = found ? found[1] : '';
      }
      var sku = abbr + (weightNum ? '-' + weightNum : '');
      sheet.getRange(i + 1, skuCol + 1).setValue(sku.toUpperCase());
      updated++;
    }
  }
  SpreadsheetApp.getUi().alert(`✅ Auto-assigned ${updated} UPPERCASE SKUs on "${sheet.getName()}".`);
}

      modified = true;
    }
  });
  if (modified || headers.length !== SHEET_HEADERS.length) {
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  }
}

// Webhook handler for incoming data
function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    Logger.log('Received webhook data: ' + JSON.stringify(data));
    
    switch(data.action) {
      case 'inventory_sync':
        return handleInventorySync(data);
      case 'shopify_order':
        return handleShopifyOrder(data);
      default:
        return forwardToInventorySystem(data);
    }
    
  } catch (error) {
    Logger.log('Error processing webhook: ' + error.toString());
    return ContentService
      .createTextOutput(JSON.stringify({status: 'error', message: error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
function handleInventorySync(data) {
  try {
    if (data.items && data.items.length > 0) {
      updateSyncSheet(data.items);
      updateInventorySheet(data.items);
      Logger.log('Successfully synced ' + data.items.length + ' items');
    }
    
    return ContentService
      .createTextOutput(JSON.stringify({status: 'ok', synced: data.items?.length || 0}))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    Logger.log('Sync error: ' + error.toString());
    throw error;
  }
}
function updateSyncSheet(items) {
  try {
    const sheet = SpreadsheetApp.openById(CONFIG.SYNC_SHEET_ID).getActiveSheet();
    
    if (sheet.getRange(1, 1).getValue() === '') {
      sheet.getRange(1, 1, 1, 6).setValues([
        ['Timestamp', 'Items Synced', 'Source', 'Status', 'Last Update', 'Notes']
      ]);
    }
    
    sheet.appendRow([
      new Date(),
      items?.length || 0,
      'Inventory System',
      'Success',
      new Date(),
      'Service Account Integration'
    ]);
    
  } catch (error) {
    Logger.log('Error updating sync sheet: ' + error.toString());
  }
}
function updateInventorySheet(items) {
  try {
    const sheet = SpreadsheetApp.openById(CONFIG.INVENTORY_SHEET_ID).getActiveSheet();
    
    if (sheet.getRange(1, 1).getValue() === '') {
      sheet.getRange(1, 1, 1, 9).setValues([
        ['ID', 'Name', 'SKU', 'Quantity', 'Base Price', 'Status', 'Category', 'Brand', 'Last Updated']
      ]);
    }
    
    if (sheet.getLastRow() > 1) {
      sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).clear();
    }
    
    if (items && items.length > 0) {
      const rows = items.map(item => [
        item.id,
        item.name,
        item.sku,
        item.quantity,
        item.basePrice,
        item.status,
        item.category,
        item.brand,
        new Date()
      ]);
      
      sheet.getRange(2, 1, rows.length, 9).setValues(rows);
    }
    
    Logger.log('Updated inventory with ' + (items?.length || 0) + ' items');
  } catch (error) {
    Logger.log('Error updating inventory: ' + error.toString());
  }
}
function forwardToInventorySystem(data) {
  try {
    const response = UrlFetchApp.fetch(CONFIG.INVENTORY_WEBHOOK_URL, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      payload: JSON.stringify({
        action: data.action || 'sync_request',
        data: data,
        timestamp: new Date().toISOString(),
        source: 'google_apps_script'
      })
    });
    
    return ContentService
      .createTextOutput(response.getContentText())
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    Logger.log('Error forwarding: ' + error.toString());
    return ContentService
      .createTextOutput(JSON.stringify({status: 'error', message: error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Orders subtraction: FlatOrderItems to Inventory
function subtractAndClearOrdersBySKU() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const orderSheet = ss.getSheetByName(ORDER_SHEET_NAME);
  const invSheet = ss.getSheetByName(SHEET_NAME);
  if (!orderSheet || !invSheet) {
    SpreadsheetApp.getUi().alert('❌ Missing FlatOrderItems or main inventory sheet.');
    return;
  }
  const orderRows = orderSheet.getDataRange().getValues();
  if (orderRows.length < 2) {
    SpreadsheetApp.getUi().alert('No orders to process.');
    return;
  }
  const orderHeaders = orderRows[0].map(h => h.toString().trim());
  const invRows = invSheet.getDataRange().getValues();
  const invHeaders = invRows[0].map(h => h.toString().trim());

  const orderSKUIdx = orderHeaders.indexOf('SKU');
  const orderQtyIdx = orderHeaders.indexOf('Quantity');
  const invSKUIdx = invHeaders.indexOf('SKU');
  const invAvailIdx = invHeaders.indexOf('Available');
  if ([orderSKUIdx, orderQtyIdx, invSKUIdx, invAvailIdx].some(i => i === -1)) {
    SpreadsheetApp.getUi().alert(
      '❌ Column index not found!\n' +
      'FlatOrderItems headers: ' + orderHeaders.join(', ') +
      '\nInventory sheet headers: ' + invHeaders.join(', ')
    );
    return;
  }

  // Map total ordered per SKU
  let skuOrderTotals = {};
  for (let i = 1; i < orderRows.length; i++) {
    const sku = (orderRows[i][orderSKUIdx] || '').toString().trim().toUpperCase();
    const qty = Number(orderRows[i][orderQtyIdx]) || 0;
    if (!sku || !qty) continue;
    skuOrderTotals[sku] = (skuOrderTotals[sku] || 0) + qty;
  }

  // Subtract from inventory
  let updated = 0;
  for (let j = 1; j < invRows.length; j++) {
    const invSKU = (invRows[j][invSKUIdx] || '').toString().trim().toUpperCase();
    if (!invSKU || skuOrderTotals[invSKU] === undefined) continue;
    let avail = Number(invRows[j][invAvailIdx]) || 0;
    let newAvail = Math.max(avail - skuOrderTotals[invSKU], 0);
    if (newAvail !== avail) updated++;
    invSheet.getRange(j + 1, invAvailIdx + 1).setValue(newAvail);
  }

  // Delete all orders except header
  if (orderSheet.getLastRow() > 1) {
    orderSheet.deleteRows(2, orderSheet.getLastRow() - 1);
  }
  SpreadsheetApp.getUi().alert(
    `✅ Inventory updated (${updated} SKUs).\nOrders sheet cleared. Ready for next batch!`
  );
}

// == Shopify Product Upload with Robust Rate-Limiting and Retry ==
function uploadProductsToShopify() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const data = sheet.getDataRange().getValues();
  const headers = data[0].map(h => h.trim());
  const rows = data.slice(1);
  const prodIdCol = headers.indexOf('Shopify Product ID');
  const varIdCol = headers.indexOf('Shopify Variant ID');
  let created = 0, failed = 0, skipped = 0;
  const logs = [];

  rows.forEach((row, idx) => {
    try {
      const rowData = {};
      headers.forEach((header, i) => { rowData[header] = row[i]; });
      // Skip if already uploaded
      if (rowData['Shopify Product ID']) { skipped++; return; }
      if (!rowData['Title']) { logs.push({row: idx + 2, status: 'SKIPPED', reason: 'Missing Title'}); skipped++; return; }

      // === Build productData object ===
      const sku = rowData['SKU'] || `SKU-${Date.now()}-${idx}`;
      const barcode = rowData['Barcode'] || `BC-${Date.now()}-${idx}`;
      const vendor = rowData['Vendor'] || 'Default Vendor';
      const productType = rowData['Type'] || '';
      const tags = rowData['Tags'] || '';
      const price = rowData['Variant Price'] || '0.00';
      const status = rowData['Status'] || 'draft';

      let productData = {
        product: {
          title: rowData['Title'],
          vendor: vendor,
          product_type: productType,
          tags: tags,
          status: status,
          variants: [{
            price: price,
            sku: sku,
            barcode: barcode,
            inventory_management: 'shopify',
            inventory_policy: 'deny',
            taxable: rowData['Variant Taxable'] === 'TRUE',
            requires_shipping: rowData['Variant Requires Shipping'] === 'TRUE',
            grams: parseFloat(rowData['Variant Grams']) || 0,
            inventory_quantity: parseInt(rowData['Variant Inventory Qty']) || 0
          }],
          images: rowData['Image Src'] ? [{ src: rowData['Image Src'] }] : []
        }
      };

      const url = `https://${SHOPIFY_STORE}/admin/api/2024-07/products.json`;
      const options = {
        method: 'post',
        contentType: 'application/json',
        headers: { 'X-Shopify-Access-Token': SHOPIFY_ACCESS_TOKEN },
        payload: JSON.stringify(productData),
        muteHttpExceptions: true
      };

      let attempts = 0, maxAttempts = 5, success = false;
      let response, responseCode, responseText;
      while (attempts < maxAttempts && !success) {
        response = UrlFetchApp.fetch(url, options);
        responseCode = response.getResponseCode();
        responseText = response.getContentText();

        if (responseCode === 201) {
          const shopifyResponse = JSON.parse(responseText);
          if (prodIdCol !== -1 && shopifyResponse.product && shopifyResponse.product.id) {
            sheet.getRange(idx + 2, prodIdCol + 1).setValue(shopifyResponse.product.id);
          }
          if (varIdCol !== -1 && shopifyResponse.product && shopifyResponse.product.variants && shopifyResponse.product.variants[0]) {
            sheet.getRange(idx + 2, varIdCol + 1).setValue(shopifyResponse.product.variants[0].id);
          }
          created++; logs.push({row: idx + 2, status: 'CREATED', id: shopifyResponse.product.id, sku: sku}); success = true;
        } else if (responseCode === 429 || responseText.indexOf('Exceeded 2 calls per second') !== -1) {
          Utilities.sleep(1200 + (attempts * 500));
          attempts++;
        } else {
          Logger.log(`Failed to create product '${rowData['Title']}': ${responseText}`);
          failed++; logs.push({row: idx + 2, status: 'FAILED', error: responseText, sku: sku}); break;
        }
      }
      Utilities.sleep(800);

    } catch (error) {
      Logger.log(`Error processing row ${idx + 2}: ${error.message}`);
      failed++; logs.push({row: idx + 2, status: 'FAILED', error: error.message});
    }
  });

  let msg = `✅ ${created} products created, ${failed} failed, ${skipped} skipped.`;
  if (failed > 0) msg += " (Check Logs)";
  SpreadsheetApp.getUi().alert(msg);

  logEvent('uploadProductsToShopify', logs);
}

// === Telegram Image Upload Helper ===
function uploadPhotoToTelegramWithMeta(base64String, meta) {
  const TELEGRAM_BOT_TOKEN = PropertiesService.getScriptProperties().getProperty('TELEGRAM_BOT_TOKEN');
  const TELEGRAM_CHAT_ID   = PropertiesService.getScriptProperties().getProperty('TELEGRAM_CHAT_ID');
  if (!TELEGRAM_BOT_TOKEN || !TELEGRAM_CHAT_ID) {
    return { ok: false, error: "Telegram token or chat_id missing." };
  }
  const matches = base64String.match(/^data:(.+);base64,(.+)$/);
  if (!matches) return { ok: false, error: "Invalid base64 string." };
  const blob = Utilities.newBlob(Utilities.base64Decode(matches[2]), matches[1], "upload.jpg");

  const caption = (meta && meta.caption) ? String(meta.caption).slice(0,1024) : "";
  const telegramUrl = `https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/sendPhoto`;
  const payload = {
    method: "post",
    payload: { chat_id: TELEGRAM_CHAT_ID, photo: blob, caption: caption }
  };
  try {
    const response = UrlFetchApp.fetch(telegramUrl, payload);
    const json = JSON.parse(response.getContentText());
    if (!json.ok) return { ok: false, error: (json.description || "Telegram API error") };
    const photoObj = json.result.photo.pop();
    const fileId = photoObj.file_id;
    const getFileUrl = `https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/getFile?file_id=${fileId}`;
    const getFileResp = UrlFetchApp.fetch(getFileUrl);
    const fileJson = JSON.parse(getFileResp.getContentText());
    if (!fileJson.ok) return { ok: false, error: (fileJson.description || "Failed to get Telegram file") };
    const filePath = fileJson.result.file_path;
    const publicUrl = `https://api.telegram.org/file/bot${TELEGRAM_BOT_TOKEN}/${filePath}`;
    return { ok: true, url: publicUrl, file_id: fileId, meta: meta || {}, telegram_message_id: json.result.message_id };
  } catch (err) {
    return { ok: false, error: err.message };
  }
}

// === Sync Inventory Availability from PriceSheet/Sheet10 ===
function syncAvailableFromSheet10() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const priceSheet = ss.getSheetByName('Sheet10');
  const shopifySheet = ss.getSheetByName(SHEET_NAME);
  if (!priceSheet || !shopifySheet) {
    SpreadsheetApp.getUi().alert('❌ One or both sheets not found!');
    return;
  }

  const priceData = priceSheet.getDataRange().getValues();
  const shopifyData = shopifySheet.getDataRange().getValues();
  if (priceData.length < 2 || shopifyData.length < 2) {
    SpreadsheetApp.getUi().alert('❌ Not enough data in one or both sheets.');
    return;
  }

  const norm = s => String(s).trim().toLowerCase();
  const pHeaders = priceData[0].map(norm);
  const sHeaders = shopifyData[0].map(norm);

  const pHandle = pHeaders.indexOf('handle');
  const pSku = pHeaders.indexOf('sku');
  const pTitle = pHeaders.indexOf('title');
  const pAvail = pHeaders.indexOf('available');
  const sHandle = sHeaders.indexOf('handle');
  const sSku = sHeaders.indexOf('sku') !== -1 ? sHeaders.indexOf('sku') : sHeaders.indexOf('variant sku');
  const sTitle = sHeaders.indexOf('title');
  const sAvail = sHeaders.indexOf('available');

  if ([pHandle, pSku, pTitle, pAvail, sHandle, sSku, sTitle, sAvail].some(idx => idx === -1)) {
    SpreadsheetApp.getUi().alert(
      '❌ Missing required columns!\n' +
      'Sheet10: ' + priceData[0].join(' | ') + '\n' +
      'ShopifyInventoryExport: ' + shopifyData[0].join(' | ')
    );
    return;
  }

  const priceLookup = {};
  for (let i = 1; i < priceData.length; i++) {
    const row = priceData[i];
    const avail = row[pAvail];
    if (row[pHandle]) priceLookup['H|' + norm(row[pHandle])] = avail;
    if (row[pSku]) priceLookup['S|' + norm(row[pSku])] = avail;
    if (row[pTitle]) priceLookup['T|' + norm(row[pTitle])] = avail;
  }

  let updated = 0, notFound = 0;
  for (let i = 1; i < shopifyData.length; i++) {
    const sRow = shopifyData[i];
    let avail = null;

    if (sRow[sHandle] && priceLookup['H|' + norm(sRow[sHandle])] !== undefined) {
      avail = priceLookup['H|' + norm(sRow[sHandle])];
    } else if (sRow[sSku] && priceLookup['S|' + norm(sRow[sSku])] !== undefined) {
      avail = priceLookup['S|' + norm(sRow[sSku])];
    } else if (sRow[sTitle] && priceLookup['T|' + norm(sRow[sTitle])] !== undefined) {
      avail = priceLookup['T|' + norm(sRow[sTitle])];
    }

    if (avail !== null && avail !== undefined && avail !== '') {
      shopifySheet.getRange(i + 1, sAvail + 1).setValue(avail);
      updated++;
    } else {
      notFound++;
    }
  }

  SpreadsheetApp.getUi().alert(`✅ Updated ${updated} rows in Inventory from Sheet10. Not found: ${notFound}`);
}

// === Move In-Stock Items to Top of Sheet ===
function moveInStockToTop() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  if (!sheet) throw new Error(`❌ Sheet "${SHEET_NAME}" not found.`);
  const data = sheet.getDataRange().getValues();
  if (data.length < 3) return;
  const headers = data[0].map(String);
  const availIdx = headers.findIndex(h => h.trim().toLowerCase() === 'available');
  if (availIdx === -1) {
    SpreadsheetApp.getUi().alert('❌ "Available" column not found!');
    return;
  }
  const header = data[0];
  const newRow = data[1];
  const rest = data.slice(2);

  const inStock = [], outOfStock = [];
  for (let i = 0; i < rest.length; i++) {
    const val = rest[i][availIdx];
    const isInStock = !isNaN(val) && Number(val) > 0;
    if (isInStock) inStock.push(rest[i]);
    else outOfStock.push(rest[i]);
  }

  const sorted = [header, newRow, ...inStock, ...outOfStock];
  sheet.clearContents();
  sheet.getRange(1, 1, sorted.length, header.length).setValues(sorted);
  SpreadsheetApp.getUi().alert(`✅ Sorted: ${inStock.length} in-stock items are now at the top.`);
}

// === Logging Helper (for debugging, monitoring, event log) ===
function logEvent(eventType, details) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let logSheet = ss.getSheetByName('EventLog');
  if (!logSheet) {
    logSheet = ss.insertSheet('EventLog');
    logSheet.appendRow(['Timestamp', 'Event Type', 'Details']);
  }
  logSheet.appendRow([new Date(), eventType, JSON.stringify(details)]);
}

// === Utility: Clear 'Processed' Flags in Orders Sheet ===
function clearProcessedFlags() {
  const orderSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ORDER_SHEET_NAME);
  const data = orderSheet.getDataRange().getValues();
  const headers = data[0];
  const processedIdx = headers.indexOf('Processed');
  if (processedIdx === -1) {
    SpreadsheetApp.getUi().alert('❌ Processed column not found.');
    return;
  }
  orderSheet.getRange(2, processedIdx + 1, orderSheet.getLastRow() - 1).clearContent();
  SpreadsheetApp.getUi().alert('✅ Cleared all processed flags.');
}

// === Reset Trigger (for auto order handling) ===
function resetTrigger() {
  const triggers = ScriptApp.getProjectTriggers();
  triggers.forEach(trigger => {
    ScriptApp.deleteTrigger(trigger);
  });
  ScriptApp.newTrigger('processOrderSheetChanges')
    .forSpreadsheet(SpreadsheetApp.getActiveSpreadsheet())
    .onChange()
    .create();
  SpreadsheetApp.getUi().alert('✅ Trigger successfully reset.');
}

// === Batch Add Items Helper (optional utility for bulk insert) ===
function batchAddItems(batchData) {
  let added = 0, errors = 0, results = [];
  batchData.forEach(data => {
    try {
      let res = addItem(data);
      if (res && res.success) {
        added++;
        results.push({row: data, status: 'success'});
      } else {
        errors++;
        results.push({row: data, status: 'fail'});
      }
    } catch (e) {
      errors++;
      results.push({row: data, status: 'fail', error: e.message});
    }
  });
  return {added, errors, results};
}
